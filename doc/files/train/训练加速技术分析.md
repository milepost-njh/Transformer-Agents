## 四、🎯 具体代码修改建议

### 修改1: 添加Flash Attention支持

```python
# 在 scaled_dot_product_attention 函数中
def scaled_dot_product_attention(q, k, v, mask=None, use_flash_attn=True):
    """
    Args:
        q: (..., seq_len_q, depth)
        k: (..., seq_len_k, depth)
        v: (..., seq_len_v, depth_v)
        mask: (..., seq_len_q, seq_len_k)
        use_flash_attn: 是否使用Flash Attention
    """
    # 尝试使用Flash Attention 2
    if use_flash_attn and torch.cuda.is_available():
        try:
            from flash_attn import flash_attn_func
            # Flash Attention要求: [B, L, H, D] 格式
            # 当前格式: [B, H, L, D]，需要转置
            if q.dtype in [torch.float16, torch.bfloat16] and mask is None:
                q_fa = q.transpose(1, 2).contiguous()  # [B, L, H, D]
                k_fa = k.transpose(1, 2).contiguous()
                v_fa = v.transpose(1, 2).contiguous()
                
                out = flash_attn_func(
                    q_fa, k_fa, v_fa,
                    dropout_p=0.0,
                    causal=False,  # 根据实际需求设置
                )
                out = out.transpose(1, 2).contiguous()  # [B, H, L, D]
                return out, None  # Flash Attention不返回权重
        except ImportError:
            pass  # 回退到标准实现
    
    # 标准实现
    matmul_qk = torch.matmul(q, k.transpose(-2, -1))
    dk = q.size()[-1]
    scaled_attention_logits = matmul_qk / torch.sqrt(torch.tensor(dk, dtype=torch.float32, device=q.device))
    
    if mask is not None:
        scaled_attention_logits = scaled_attention_logits.masked_fill(mask == 1, -1e9)
    
    attention_weights = F.softmax(scaled_attention_logits, dim=-1)
    output = torch.matmul(attention_weights, v)
    
    return output, attention_weights
```

### 修改2: 添加梯度累积

```python
def train_model(
        epochs: int,
        model: torch.nn.Module,
        optimizer: torch.optim.Optimizer,
        train_loader: torch.utils.data.DataLoader,
        ...,
        gradient_accumulation_steps: int = 1,  # 新增参数
):
    ...
    for epoch in range(epochs):
        for batch_idx, batch in enumerate(train_loader):
            # 计算损失但不立即更新
            loss_val, acc_val = train_step(
                batch=batch, 
                transformer=model, 
                optimizer=optimizer,
                scheduler=None,  # scheduler在累积完成后更新
                device=device,
                moe_config=moe_config,
                update_weights=False,  # 新增标志
            )
            
            # 累积梯度
            loss_val = loss_val / gradient_accumulation_steps
            loss_val.backward()
            
            # 只在累积完成后更新权重
            if (batch_idx + 1) % gradient_accumulation_steps == 0:
                # 梯度裁剪
                grad_norm = torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=0.5)
                
                # 更新权重
                optimizer.step()
                optimizer.zero_grad(set_to_none=True)
                
                if scheduler is not None:
                    scheduler.step()
                
                global_step += 1
```

class EncoderLayer(nn.Module):

                x, src_mask, past_key_value, use_cache,
